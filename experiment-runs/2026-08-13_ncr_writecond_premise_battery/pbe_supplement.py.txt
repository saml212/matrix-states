#!/usr/bin/env python3
"""Premise battery SUPPLEMENT (coordinator, 2026-08-13): the audited cell-2
spec used eval_both_arms, which evaluates the runner's default hops (1,2,3)
and takes no hops argument -- leaving the pre-registered h=61 coordinates
(c and d of the AC/BD grid) unmeasured. This supplement applies the SAME
instrument P1a used (eval_arm_at_hops, identical signature and seed) to the
RESTORED trained arms, both write modes, at the registered hop set.
Disclosed spec gap; no new instrument, no new knobs."""
import json, os, sys, time
import torch
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import ncr_lm_wave1_runner as R

BASE_SEED = 90210
HOPS = (1, 13, 37, 61)
CKPT = os.path.expanduser("~/ncr_g3b31_contrastive/results/mob_g3b31_compB_s0_ckpts/mob_g3b31_compB_s0.ckpt.pt")
OUT = os.path.expanduser("~/ncr_writecond/results/writecond_premise_SUPP.json")

def main():
    t0 = time.time()
    device = "cuda"
    pools, cfg, pool_report = R.build_grammar_pools_and_cfg(seed=0)
    pools = pools.to(device)
    ckpt = R.load_checkpoint(CKPT, device)
    assert ckpt is not None, f"checkpoint missing at {CKPT}"
    arms, opts, data_gen = R.restore_arms_and_opts(ckpt, pool_report["vocab_size_total"],
                                                    lr=3e-4, device=device, freeze_entity_adapter=False)
    with torch.no_grad():
        p1b_deep = R.eval_arm_at_hops(arms["full_graft"], pools, cfg, HOPS, 256, device,
                                       BASE_SEED, read_ablate=False, teacher_force=True)
        p0_deep = R.eval_arm_at_hops(arms["full_graft"], pools, cfg, HOPS, 256, device,
                                      BASE_SEED, read_ablate=False, teacher_force=False)
    rec = dict(cell="SUPP", ckpt_step=ckpt["step"], n=256,
               P1b_deep=dict(teacher_force=True, result=p1b_deep),
               P0_deep=dict(teacher_force=False, result=p0_deep),
               elapsed_s=time.time() - t0,
               note="disclosed spec-gap supplement; same instrument as P1a")
    os.makedirs(os.path.dirname(OUT), exist_ok=True)
    with open(OUT, "w") as f:
        json.dump(rec, f, indent=2, default=str)
    print("SUPP_DONE")

if __name__ == "__main__":
    main()
